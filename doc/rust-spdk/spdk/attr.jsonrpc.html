Attribute Macro jsonrpc 

#[jsonrpc]
Available on crate feature json-rpc only.
Expand description

Marks a standalone function as callable through the SPDK JSON-RPC server.

The attributed function may be synchronous or asynchronous and take zero or more parameters with types implementing the Deserialize trait. It may also return the unit type, any type implementing the Serialize trait, or a Result<T, E> object where the error type, E, is convertible to an RpcError object through the From trait.

§Example

The following example demonstrates both synchronous and asynchronous JSON-RPC callable functions.

use std::io;

use async_fs::File;
use futures::io::AsyncWriteExt;

use spdk::{self, env::CpuSet, errors::ERANGE, jsonrpc, runtime, thread};

/// Synchronously prints a greeting.
#[jsonrpc]
fn hello_world() {
    println!("HELLO, WORLD!");
}

/// Synchronously prints the message from the requestor.
#[jsonrpc]
fn print_message(msg: &str) {
    println!("MESSAGE: {}", msg);
}

/// Asynchronously adds two integers and returns the result.
#[jsonrpc]
async fn add_two(a: i32, b: i32) -> i32 {
    thread::spawn_local(async { a + b }).await
}

/// Plays a guessing game asynchronously with the requestor.
#[jsonrpc]
async fn guess_number(guess: u8) -> spdk::Result<String> {
    thread::spawn(c"guess_number", &CpuSet::any(), || async {
        if !(1..=10).contains(&guess) {
            return Err(ERANGE);
        }

        if guess == 5 {
            return Ok("CORRECT!".to_string());
        }

        Ok("WRONG!".to_string())
    })?
    .await
}

/// Writes the specified text into a file.
#[jsonrpc]
async fn write_file(filename: &str, text: &str) -> io::Result<()> {
    let mut file = File::create(filename).await?;

    file.write_all(text.as_bytes()).await
}

#[spdk::main]
async fn main() {
    runtime::wait_for_shutdown().await;
}
