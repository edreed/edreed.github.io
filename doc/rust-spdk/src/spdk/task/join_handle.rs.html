14/// A virtual function table (vtable) that specifies the operations that can be performed on a
15/// [`RawJoinHandle`].
16///
17/// The pointer passed to all functions in this vtable is the `data` pointer of the enclosing
18/// [`RawJoinHandle`] object. The vtable is used to construct a [`RawJoinHandle`] that is embedded
19/// in a [`JoinHandle`]. The vtable is used by `JoinHandle` to orchestrate receiving the result of
20/// an asynchronous operation.
21pub(crate) struct RawJoinHandleVTable<R>
22where
23    R: 'static,
24{
25    /// This function is called when a [`JoinHandle`] is polled through its [`Future`] trait
26    /// implementation. It returns a [`Poll<T>`] value indicating whether the task has completed
27    /// and, if so, the result of the task.
28    pub(crate) poll_result: unsafe fn(*const (), &mut Context<'_>) -> Poll<R>,
29
30    /// This function is called when a [`JoinHandle`] is dropped. It should perform any necessary
31    /// cleanup for the task.
32    pub(crate) drop: unsafe fn(*mut ()),
33}
34
35/// A raw handle to a task that can be used to await the result of the task.
36pub(crate) struct RawJoinHandle<R>
37where
38    R: 'static,
39{
40    vtable: &'static RawJoinHandleVTable<R>,
41    data: *mut (),
42}

44/// A handle that awaits the result of a task.
45///
46/// Dropping a [`JoinHandle`] will detach the task leaving no way to join on
47/// it or obtain its result.
48///
49/// A [`JoinHandle`] is created when a task is spawned.
50#[must_use = "this JoinHandle must be awaited to ensure proper lifetime management"]
51pub struct JoinHandle<'a, F, R>
52where
53    F: Future<Output = R> + 'a,
54    R: 'static,
55{
56    raw: RawJoinHandle<R>,
57    _task: PhantomData<&'a F>,
58}
59
60impl<'a, F, R> JoinHandle<'a, F, R>
61where
62    F: Future<Output = R> + 'a,
63    R: 'static,
64{
65    /// Creates a new `JoinHandle` with the specified `data` pointer and `vtable`.
66    const unsafe fn new(data: *mut (), vtable: &'static RawJoinHandleVTable<R>) -> Self {
67        Self {
68            raw: RawJoinHandle { vtable, data },
69            _task: PhantomData,
70        }
71    }
72
73    /// Creates a `JoinHandle` from a local task reference counted by `Rc`.
74    pub(crate) fn from_local_task<T>(rc: Rc<T>) -> Self
75    where
76        T: RcTask<Output = R>,
77    {
78        let vtable = local_task::join_handle_vtable::<T>();
79        let data = Rc::into_raw(rc).cast_mut() as *mut _;
80
81        unsafe { Self::new(data, vtable) }
82    }
83
84    /// Creates a `JoinHandle` from a remote task reference counted by `Arc`.
85    pub(crate) fn from_remote_task<T>(arc: Arc<T>) -> Self
86    where
87        T: ArcTask<Output = R>,
88    {
89        let vtable = remote_task::join_handle_vtable::<T>();
90        let data = Arc::into_raw(arc).cast_mut() as *mut _;
91
92        unsafe { Self::new(data, vtable) }
93    }
94}

96impl<'a, F, R> Future for JoinHandle<'a, F, R>
97where
98    F: Future<Output = R> + 'a,
99    R: 'static,
100{
101    type Output = R;
102
103    fn poll(self: std::pin::Pin<&mut Self>, cx: &mut Context<'_>) -> Poll<Self::Output> {
104        unsafe { (self.raw.vtable.poll_result)(self.raw.data, cx) }
105    }
106}
