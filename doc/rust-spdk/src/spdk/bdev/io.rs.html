99impl From<spdk_bdev_io_type> for IoType {
100    fn from(value: spdk_bdev_io_type) -> Self {
101        if value as u32 >= SPDK_BDEV_IO_TYPE_INVALID as u32
102            || value as u32 <= SPDK_BDEV_IO_TYPE_WRITE_UNCORRECTABLE as u32
103        {
104            // SAFETY: The specified value is within the correct range for direct transmutation.
105            return unsafe { transmute::<u8, Self>(value as u8) };
106        }
107
108        Self::Invalid
109    }
110}
111
112impl From<u8> for IoType {
113    fn from(value: u8) -> Self {
114        if value >= Self::Invalid as u8 || value <= Self::WriteUncorrectable as u8 {
115            // SAFETY: The specified value is within the correct range for direct transmutation.
116            return unsafe { transmute::<u8, Self>(value) };
117        }
118
119        Self::Invalid
120    }
121}

123impl From<IoType> for spdk_bdev_io_type {
124    fn from(value: IoType) -> Self {
125        // SAFETY: `IoType` has a 1:1 mapping to `spdk_bdev_io_type` values.
126        unsafe { transmute(value as u32) }
127    }

130/// An error describing the reason for a BDev I/O failure.
131#[derive(Copy, Clone, Debug, Error)]
132pub enum IoError {
133    /// A general error occurred.
134    ///
135    /// The `Errno` tuple field contains a Linux error code describing the reason.
136    #[error(transparent)]
137    GeneralError(Errno),
138
139    /// The I/O was aborted.
140    #[error("I/O aborted")]
141    Aborted,
142
143    /// The first fused request in a compare-and-write operation failed.
144    #[error("the first fused request failed")]
145    FirstFusedFailed,
146
147    /// The block data in a compare or compare-and-write operation has changed.
148    #[error("block data has changed")]
149    Miscompare,
150
151    /// There are currently no resources to submit a request.
152    ///
153    /// The request should be retried later when resources become available.
154    #[error("out of resources")]
155    NoMem,
156
157    /// A SCSI error occurred.
158    ///
159    /// The `ScsiStatus` tuple field contains the SCSI status information.
160    #[cfg(feature = "scsi")]
161    #[error("a SCSI error occurred: {0}")]
162    ScsiError(ScsiStatus),
163
164    /// An NVMe error occurred.
165    ///
166    /// The `NvmeStatus` tuple field contains the NVMe status information.
167    #[cfg(feature = "nvmf")]
168    #[error("an NVME error occurred: {0}")]
169    NvmeError(NvmeStatus),
170
171    /// A general I/O failure occurred.
172    #[error("I/O failed")]
173    Failed,
174
175    /// The I/O is still pending completion.
176    #[error("I/O pending")]
177    Pending,
178}
179
180// If this assertion failes, a new `spdk_bdev_io_status` may have been added and `IoError` should be
181// updated accordingly.
182const_assert_eq!(
183    SPDK_BDEV_IO_STATUS_AIO_ERROR as i32,
184    spdk_bdev_io_status::SPDK_MIN_BDEV_IO_STATUS as i32
185);

187impl From<Errno> for IoError {
188    fn from(value: Errno) -> Self {
189        match value {
190            ENOMEM => Self::NoMem,
191            ECANCELED => Self::Aborted,
192            EINPROGRESS => Self::Pending,
193            _ => Self::GeneralError(value),
194        }
195    }
196}
