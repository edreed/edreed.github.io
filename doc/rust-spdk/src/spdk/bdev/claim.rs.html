20/// Represents the type of claim on a block device.
21#[derive(Clone, Debug, PartialEq, Eq)]
22pub enum ClaimType<'a> {
23    /// Exclusive write claim. Only the claimant can have write access to the block device.
24    ///
25    /// The `String` parameter represents the name of the claimant and is used for logging purposes.
26    /// It must be 31 characters or less.
27    ReadManyWriteOne(&'a CStr),
28
29    /// Read many, write none claim. Multiple entities can have read access, but no entity can have
30    /// write access to the block device.
31    ///
32    /// The `String` parameter represents the name of the claimant and is used for logging purposes.
33    /// It must be 31 characters or less.
34    ReadManyWriteNone(&'a CStr),
35
36    /// Read many, write many shared claim. Multiple entities using the same shared claim key can
37    /// have read and write access to the block device.
38    ///
39    /// The `String` parameter represents the name of the claimant and is used for logging purposes.
40    /// It must be 31 characters or less.
41    ///
42    /// The `u64` parameter represents the shared claim key.
43    ReadManyWriteShared(&'a CStr, u64),
44}
45
46impl<'a> ClaimType<'a> {
47    /// Converts the `ClaimType` into the corresponding SPDK claim type and claim options.
48    pub(crate) fn into_params(self) -> Result<(spdk_bdev_claim_type, spdk_bdev_claim_opts)> {
49        let (claim_type, name, shared_claim_key) = match self {
50            ClaimType::ReadManyWriteOne(name) => (SPDK_BDEV_CLAIM_READ_MANY_WRITE_ONE, name, 0),
51            ClaimType::ReadManyWriteNone(name) => (SPDK_BDEV_CLAIM_READ_MANY_WRITE_NONE, name, 0),
52            ClaimType::ReadManyWriteShared(name, key) => {
53                (SPDK_BDEV_CLAIM_READ_MANY_WRITE_SHARED, name, key)
54            }
55        };
56
57        let mut opts = MaybeUninit::uninit();
58
59        unsafe {
60            spdk_bdev_claim_opts_init(opts.as_mut_ptr(), size_of::<spdk_bdev_claim_opts>());
61        }
62
63        let mut opts = unsafe { opts.assume_init() };
64
65        let mut claim_name = Cursor::new(unsafe {
66            slice::from_raw_parts_mut(opts.name.as_mut_ptr() as *mut u8, opts.name.len())
67        });
68
69        claim_name
70            .write_all(name.to_bytes_with_nul())
71            .map_err(|_| EINVAL)?;
72
73        opts.shared_claim_key = shared_claim_key;
74
75        Ok((claim_type, opts))
76    }
77}
