18/// Represents the type of claim on a block device.
19#[derive(Clone, Debug, PartialEq, Eq)]
20pub enum ClaimType<'a> {
21    /// Exclusive write claim. Only the claimant can have write access to the block device.
22    ///
23    /// The `String` parameter represents the name of the claimant and is used for logging purposes.
24    /// It must be 31 characters or less.
25    ReadManyWriteOne(&'a CStr),
26
27    /// Read many, write none claim. Multiple entities can have read access, but no entity can have
28    /// write access to the block device.
29    ///
30    /// The `String` parameter represents the name of the claimant and is used for logging purposes.
31    /// It must be 31 characters or less.
32    ReadManyWriteNone(&'a CStr),
33
34    /// Read many, write many shared claim. Multiple entities using the same shared claim key can
35    /// have read and write access to the block device.
36    ///
37    /// The `String` parameter represents the name of the claimant and is used for logging purposes.
38    /// It must be 31 characters or less.
39    ///
40    /// The `u64` parameter represents the shared claim key.
41    ReadManyWriteShared(&'a CStr, u64),
42}
43
44impl<'a> ClaimType<'a> {
45    /// Converts the `ClaimType` into the corresponding SPDK claim type and claim options.
46    pub(crate) fn into_params(self) -> Result<(spdk_bdev_claim_type, spdk_bdev_claim_opts)> {
47        let (claim_type, name, shared_claim_key) = match self {
48            ClaimType::ReadManyWriteOne(name) => (SPDK_BDEV_CLAIM_READ_MANY_WRITE_ONE, name, 0),
49            ClaimType::ReadManyWriteNone(name) => (SPDK_BDEV_CLAIM_READ_MANY_WRITE_NONE, name, 0),
50            ClaimType::ReadManyWriteShared(name, key) => {
51                (SPDK_BDEV_CLAIM_READ_MANY_WRITE_SHARED, name, key)
52            }
53        };
54
55        let mut opts = MaybeUninit::uninit();
56
57        unsafe {
58            spdk_bdev_claim_opts_init(opts.as_mut_ptr(), size_of::<spdk_bdev_claim_opts>());
59        }
60
61        let mut opts = unsafe { opts.assume_init() };
62
63        let mut claim_name = Cursor::new(unsafe {
64            slice::from_raw_parts_mut(opts.name.as_mut_ptr() as *mut u8, opts.name.len())
65        });
66
67        claim_name
68            .write_all(name.to_bytes_with_nul())
69            .map_err(|_| EINVAL)?;
70
71        opts.shared_claim_key = shared_claim_key;
72
73        Ok((claim_type, opts))
74    }
75}
