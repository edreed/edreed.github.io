37/// A type alias for the promissory that receives the buffer allocation from a call to the
38/// [`IoRequest<T>::allocate_buffers`] method.
39type AllocateBuffersPromissory<'a, T> = Promissory<(), IoError, PhantomData<&'a mut IoRequest<T>>>;

41/// Represents driver-specific context for an I/O request.
42///
43/// The type parameter `T` is the I/O context type for the BDev implementation.
44#[derive(Default)]
45pub(super) struct IoRequestCtx<'a, T>
46where
47    T: Default + 'static,
48{
49    buf_promissory: Option<Weak<AllocateBuffersPromissory<'a, T>>>,
50    inner: T,
51}

53/// A BDev I/O request.
54///
55/// The type parameter `T` is the I/O context type for the BDev implementation.
56#[repr(transparent)]
57pub struct IoRequest<T>
58where
59    T: Default + 'static,
60{
61    io: NonNull<spdk_bdev_io>,
62    _ctx: PhantomData<T>,
63}
64
65impl<T> IoRequest<T>
66where
67    T: Default + 'static,

69    /// Initializes a newly submitted I/O request.
70    ///
71    /// # Safety
72    ///
73    /// This function must only be called from the I/O submission callback to initialize a newly
74    /// submitted I/O request. It initializes the driver's I/O context to a default value.
75    pub(super) unsafe fn new_in_place(io: *mut spdk_bdev_io) -> Self {
76        unsafe {
77            (*io)
78                .driver_ctx
79                .as_mut_ptr()
80                .cast::<IoRequestCtx<T>>()
81                .write(Default::default());
82
83            // SAFETY: This type is a transparent wrapper around `NonNull<spdk_bdev_io>`.
84            transmute(NonNull::new_unchecked(io))
85        }
86    }
87    /// Returns the raw pointer to the I/O request.
88    pub(crate) fn as_ptr(&self) -> *mut spdk_bdev_io {
89        self.io.as_ptr()
90    }

93    pub(crate) fn bdev(&self) -> *mut spdk_bdev {
94        unsafe { self.io.as_ref().bdev }

97    /// Returns the type of the I/O request.
98    pub fn io_type(&self) -> IoType {
99        (unsafe { self.io.as_ref().type_ }).into()

102    /// Returns the thread associated with the I/O request. The I/O request must be completed on
103    /// this thread.
104    pub fn thread(&self) -> Thread<thread::OwnedBy<'_, Self>> {
105        // SAFETY: The thread associated with the I/O request is guaranteed to be non-null and
106        // valid for the lifetime of the I/O request.
107        unsafe { Thread::with_owner(self, spdk_bdev_io_get_thread(self.as_ptr())) }
108    }

110    /// Returns the block device associated with the I/O request.

112        // SAFETY: The block device associated with the I/O request is guaranteed to be non-null and
113        // valid for the lifetime of the I/O request.
114        unsafe { BDev::with_owner(self, self.io.as_ref().bdev) }
115    }

117    /// Returns the buffers associated with the I/O request.
118    pub fn buffers(&self) -> &[IoSlice<'_>] {
119        unsafe {
120            let mut iovecs = ptr::null_mut();

122
123            spdk_bdev_io_get_iovec(self.as_ptr(), &mut iovecs, &mut iovec_count);
124
125            slice::from_raw_parts(iovecs as *const IoSlice<'_>, iovec_count as usize)
126        }
127    }

138        }

177    /// Invoked when buffers requested by [`IoRequest<T>::allocate_buffers`] have been allocated for
178    /// the I/O request.
179    ///
180    /// [`IoRequest<T>::allocate_buffers`]: method@IoRequest::allocate_buffers
181    unsafe extern "C" fn buffers_allocated(
182        _ch: *mut spdk_io_channel,
183        io: *mut spdk_bdev_io,
184        success: bool,
185    ) {
186        let mut io: Self = Self {
187            io: unsafe { NonNull::new_unchecked(io) },
188            _ctx: PhantomData,
189        };
190        let p = io
191            .internal_ctx_mut()
192            .buf_promissory
193            .take()
194            .expect("promissory present")
195            .upgrade()
196            .expect("promissory has strong references");
197
198        Promissory::set_result(p, if_else!(success, Ok(()), Err(EINVAL.into())));
199    }

201    /// Allocates buffers aligned to the BDev's requirement for the I/O request.
202    ///
203    /// Allocation will only occur if no buffers are assigned or the buffers are not aligned to the
204    /// BDev's requirement. If the buffers are not aligned, this call will cause a copy from the
205    /// current buffers to a bounce buffer on write or a copy from the bounce buffer to the current
206    /// buffers on read.
207    ///
208    /// If no buffers are currently assigned to this I/O request, the `length` parameter specifies
209    /// the size of the buffers to allocate in bytes. This value must be no larger than
210    /// `SPDK_BDEV_LARGE_BUF_MAX_SIZE`.
211    ///
212    /// Any buffers allocated by this method will automatically be freed on completion of this I/O
213    /// request.
214    pub async fn allocate_buffers<'a>(&'a mut self, length: u64) -> IoResult<()> {
215        Promise::with_context(PhantomData::<&'a mut Self>)
216            .request(move |p| {
217                self.internal_ctx_mut()
218                    .buf_promissory
219                    .replace(Rc::downgrade(p));
220
221                unsafe {
222                    spdk_bdev_io_get_buf(self.as_ptr(), Some(Self::buffers_allocated), length)
223                };
224
225                Poll::Pending
226            })
227            .await
228    }

230    /// Completes the I/O request with the specified status.
231    ///
232    /// # Safety
233    ///
234    /// This method may only be called on the thread used to submit the I/O operation.
235    ///
236    /// # Panics
237    ///
238    /// This method panics on debug builds if not called on the thread associated with the I/O.
239    pub(super) unsafe fn complete(mut self, status: IoResult<()>) {
240        debug_assert!(self.thread().is_current());
241
242        let io_status = match status {
243            Ok(_) => SPDK_BDEV_IO_STATUS_SUCCESS,
244            Err(e) => match e {
245                IoError::GeneralError(errno) => unsafe {
246                    spdk_bdev_io_set_aio_status(self.as_ptr(), -errno.errno())
247                },
248                IoError::Aborted => SPDK_BDEV_IO_STATUS_ABORTED,
249                IoError::FirstFusedFailed => SPDK_BDEV_IO_STATUS_FIRST_FUSED_FAILED,
250                IoError::Miscompare => SPDK_BDEV_IO_STATUS_MISCOMPARE,
251                IoError::NoMem => SPDK_BDEV_IO_STATUS_NOMEM,
252                #[cfg(feature = "scsi")]
253                IoError::ScsiError(status) => unsafe {
254                    spdk_bdev_io_set_scsi_status(
255                        self.as_ptr(),
256                        status.sc.into(),
257                        status.sk.into(),
258                        status.asc,
259                        status.ascq,
260                    )
261                },
262                #[cfg(feature = "nvmf")]
263                IoError::NvmeError(status) => unsafe {
264                    spdk_bdev_io_set_nvme_status(
265                        self.as_ptr(),
266                        status.cdw0,
267                        status.sct.into(),
268                        status.sc.into(),
269                    )
270                },
271                IoError::Pending => panic!("I/O completed with pending status"),
272                _ => SPDK_BDEV_IO_STATUS_FAILED,
273            },
274        };
275
276        unsafe {
277            ptr::drop_in_place(self.io.as_mut().driver_ctx.as_mut_ptr() as *mut IoRequestCtx<T>);
278
279            spdk_bdev_io_complete(self.as_ptr(), io_status);
280        }
281    }
