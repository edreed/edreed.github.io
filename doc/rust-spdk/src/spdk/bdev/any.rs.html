11/// A placeholder type used with [`BDev`] that represents any block device.
12///
13/// This type is used as a generic placeholder for any block device, without taking ownership of it.
14/// In order to keep the `spdk_bdev` alive for the lifetime of the `Any` instance, the constructor
15/// functions open a read-only descriptor to the block device and stores it in the returned
16/// instance. This may fail if opening the read-only descriptor to the block device fails. Users of
17/// this object should be prepared to handle these errors and fail gracefully.
18///
19/// [`BDev`]: super::BDev
20#[repr(transparent)]
21pub struct Any(NonNull<spdk_bdev_desc>);
22
23unsafe impl Send for Any {}
24unsafe impl Sync for Any {}
25
26impl Any {
27    /// Create an `Any` instance from a raw `spdk_bdev` pointer without checking if it is null.
28    ///
29    /// # Returns
30    ///
31    /// This method returns an error if it fails to open a descriptor on the block device, and
32    /// `Ok(any)` otherwise.
33    ///
34    /// # Safety
35    ///
36    /// `bdev` must be non-null and a pointer to a valid `spdk_bdev` structure.
37    pub(crate) unsafe fn from_ptr_unchecked(bdev: *mut spdk_bdev) -> Result<Self> {
38        Self::with_name(unsafe { CStr::from_ptr(spdk_bdev_get_name(bdev)) })
39    }
40
41    /// Create an `Any` instance from the name of a block device.
42    ///
43    /// # Returns
44    ///
45    /// This method returns `Ok(any)` if a block device with the given name exists and a descriptor
46    /// can be successfully opened. It returns an error if it fails to open a descriptor on
47    /// the block device.
48    pub(crate) fn with_name(name: &CStr) -> Result<Self> {
49        let desc = open_desc(name, false)?;
50
51        // SAFETY: This type is a transparent wrapper around `NonNull<spdk_bdev_desc>`.
52        Ok(unsafe { transmute::<NonNull<spdk_bdev_desc>, Self>(desc) })
53    }
54}
55
56impl Drop for Any {
57    fn drop(&mut self) {
58        // SAFETY: The pointer is guaranteed to be valid as long as the Any instance exists.
59        unsafe { spdk_bdev_close(self.0.as_ptr()) };
60    }
61}
62
63impl AsRawBDev for Any {
64    fn as_raw_bdev(&self) -> *mut spdk_bdev {
65        unsafe { spdk_bdev_desc_get_bdev(self.0.as_ptr()) }
66    }
