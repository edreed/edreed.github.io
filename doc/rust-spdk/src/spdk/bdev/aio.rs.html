23/// A Linux AIO-based BDev providing block layer access to a backing file or block device.
24///
25/// # Examples
26///
27/// ```no_run
28#[doc = include_str!("../../examples/bdev_aio.rs")]
29/// ```
30#[repr(transparent)]
31pub struct Aio(NonNull<spdk_bdev>);
32
33unsafe impl Send for Aio {}
34
35impl Aio {
36    /// Create a new AIO block device.
37    ///
38    /// # Parameters
39    ///
40    /// - `name`: The name of the AIO block device.
41    /// - `filename`: The path to the backing file or block device.
42    /// - `block_size`: The block size in bytes. If not specified, the block size will be detected
43    ///   from the device specified in `filename`.
44    /// - `readonly`: Whether the file or device should be opened read-only.
45    /// - `falloc`: Whether to preallocate the file.
46    /// - `uuid`: The UUID of the device. If not specifed, a unique UUID will be generated.
47    /// - `nowait`: If `true`, do not wait if I/O would block for any reason. This may only be
48    ///   applied to block devices.
49    ///
50    /// # Returns
51    ///
52    /// Returns a `Result` containing a [`BDev<Aio>`] instance that owns the newly created BDev if successful.
53    pub async fn new<P: AsRef<Path>>(
54        name: &CStr,
55        filename: P,
56        block_size: Option<u32>,
57        readonly: bool,
58        falloc: bool,
59        uuid: Option<Uuid>,
60        nowait: bool,
61    ) -> Result<BDev<Aio>> {
62        let filename =
63            CString::new(filename.as_ref().as_os_str().as_bytes()).map_err(|_| EINVAL)?;
64
65        unsafe {
66            to_result!(create_aio_bdev(
67                name.as_ptr(),
68                filename.as_ptr(),
69                block_size.unwrap_or(0),
70                readonly,
71                falloc,
72                uuid.as_ref().map(Uuid::as_ptr).unwrap_or(null()),
73                nowait
74            ))?
75        }
76
77        wait_for_examination().await?;
78
79        let desc = NonNull::new(unsafe { spdk_bdev_get_by_name(name.as_ptr()) }).ok_or(EINVAL)?;
80
81        // SAFETY: `Aio` is a transparent wrapper around `NonNull<spdk_bdev>`, and `desc` is
82        // guaranteed to be non-null.
83        Ok(unsafe { transmute::<NonNull<spdk_bdev>, BDev<Aio>>(desc) })
84    }
85}

87impl AsRawBDev for Aio {
88    fn as_raw_bdev(&self) -> *mut spdk_bdev {
89        self.0.as_ptr()
90    }
91}

93impl OwnedOps for Aio {
94    async fn destroy(self) -> Result<()> {
95        Promise::new()
96            .request(move |p| {
97                let (cb_fn, cb_arg) = Promissory::callback_with_status(p);
98
99                unsafe {
100                    bdev_aio_delete(
101                        spdk_bdev_get_name(self.as_raw_bdev()),
102                        Some(cb_fn),
103                        cb_arg.cast_mut() as *mut _,
104                    );
105                }
106
107                Poll::Pending
108            })
109            .await
110    }
111}
112
113impl From<Owned> for Aio {
114    fn from(owned: Owned) -> Self {
115        // SAFETY: We are creating an `Aio` instance from an `Owned` device, which guarantees that
116        // the underlying `spdk_bdev` pointer is valid and uniquely owned.
117        Self(unsafe { NonNull::new_unchecked(owned.as_raw_bdev()) })
118    }
