1//! Infrastructure and abstractions supporting asynchronous execution on SPDK threads.
2//!
3//! An SPDK thread is a lightweight, stackless thread of execution and does not necessarily
4//! correspond 1:1 with a posix thread. Instead, a lower-level framework like the [SPDK Event
5//! Framework] polls each SPDK thread for work. This allows the SPDK Event Framework to multiplex
6//! many SPDK threads on a smaller number of posix threads. In this crate, the SPDK thread is
7//! implemented by the [`Thread`] type.
8//!
9//! There are two mechanisms for scheduling work on an SPDK thread: messages and pollers. A message
10//! consists of a function object sent via the [`Thread::send_msg`] function. A poller is a type
11//! implementing the [`Polled`] trait that has its [`Polled::poll`] method called periodically.
12//!
13//! See [Message Passing and Concurrency] for more details on the SPDK threading model.
14//!
15//! [SPDK Event Framework]: https://spdk.io/doc/event.html
16//! [Message Passing and Concurrency]: https://spdk.io/doc/concurrency.html

59/// A trait for SPDK thread types that provides access to the underlying raw `spdk_thread` pointer.
60pub trait AsRawThread {
61    /// Returns a raw pointer to the underlying `spdk_thread` structure.
62    fn as_raw_thread(&self) -> *mut spdk_thread;
63}

65/// An abstraction of a lightweight, stackless thread of execution.
66///
67/// The type `T` determines how the thread's ownership is managed.
68///
69/// `Thread<Owned>` represents an owned thread where the thread's lifetime is scoped to the
70/// [`Owned`] instance.
71///
72/// `Thread<Unowned>` represents a reference to a thread where the its lifetime is not managed by
73/// the [`Unowned`] instance. Additional care must be used with an `Unowned` thread to ensure that
74/// the SPDK thread remains valid for the duration of the `Thread<Unowned>` instance.
75///
76/// `Thread<OwnedBy<'a, O>>` represents a thread that is owned by another entity, `O` with lifetime
77/// `'a`. The thread's lifetime is tied to the lifetime of the owner, and care must be taken to
78/// ensure that the owner outlives the thread.
79///
80/// `Thread<App>` represents the SPDK application thread. This thread lives for the duration of the
81/// `main` function.
82#[repr(transparent)]
83pub struct Thread<T>(T)
84where
85    T: AsRawThread;
86
87unsafe impl<T> Send for Thread<T> where T: AsRawThread + Send {}
88unsafe impl<T> Sync for Thread<T> where T: AsRawThread + Sync {}
89
90impl<T> Thread<T>
91where
92    T: AsRawThread,
93{
94    /// Returns a pointer to the underlying `spdk_thread` structure.
95    fn as_ptr(&self) -> *mut spdk_thread {
96        self.0.as_raw_thread()
97    }
98
99    /// Returns an unowned reference to this thread.
100    ///
101    /// # Safety
102    ///
103    /// The caller must ensure that the returned `Thread<Unowned>` instance is not used after the SPDK
104    /// thread becomes invalid.
105    pub(crate) unsafe fn as_unowned(&self) -> Thread<Unowned> {
106        Thread(unsafe { Unowned::new_unchecked(self.as_ptr()) })
107    }
108
109    /// Returns the name of this thread.
110    pub fn name(&self) -> &CStr {
111        unsafe {
112            let name = spdk_thread_get_name(self.as_ptr());
113
114            CStr::from_ptr(name)
115        }
116    }
117
118    /// Returns the unique identifier for this thread.
119    pub fn id(&self) -> u64 {
120        unsafe { spdk_thread_get_id(self.as_ptr()) }
121    }
122
123    /// Bind or unbind the thread to its current CPU core.
124    pub fn bind(&self, bind: bool) {
125        unsafe { spdk_thread_bind(self.as_ptr(), bind) }
126    }
127
128    /// Returns whether this thread is the current thread.
129    pub fn is_current(&self) -> bool {
130        try_with_current(|current| self.as_ptr() == current.as_ptr()).unwrap_or(false)
131    }
132
133    /// Returns whether the thread is bound to its current CPU core.
134    pub fn is_bound(&self) -> bool {
135        unsafe { spdk_thread_is_bound(self.as_ptr() as *mut _) }
136    }
137
138    /// Returns the CPU set for this thread.
139    pub fn cpuset(&self) -> CpuSet {
140        let mut cpuset = CpuSet::empty();
141
142        unsafe {
143            spdk_cpuset_copy(
144                cpuset.as_mut_ptr(),
145                spdk_thread_get_cpumask(self.as_ptr() as *mut _),
146            )
147        };
148
149        cpuset
150    }
151
152    /// Invokes a function sent via [`Thread::send_msg`] on this [`Thread`]`.
153    unsafe extern "C" fn handle_msg<F>(ctx: *mut c_void)
154    where
155        F: FnOnce(),
156    {
157        let msg_fn = unsafe { Box::from_raw(ctx as *mut F) };
158
159        (*msg_fn)();
160    }
161
162    /// Sends a message function to be executed on this [`Thread`].
163    ///
164    /// The message is sent asynchronously. This function may return before the message function is
165    /// called.
166    ///
167    /// # Return
168    ///
169    /// This function returns `Ok(())` if the message function was successfully queued.
170    ///
171    /// This function returns `Err(`[`ENOMEM`]`)` if the message could not be allocated and
172    /// `Err(`[`EIO`]`)` if the message could not be sent to the destination thread.
173    ///
174    /// # Examples
175    ///
176    /// ```no_run
177    /// use spdk::thread::Thread;
178    ///
179    /// thread::with_current(|current| {
180    ///     assert!(current.send_msg(|| println!("Hello, World!")).is_ok());
181    /// });
182    /// ```
183    ///
184    /// [`EIO`]: crate::errors::EIO
185    /// [`ENOMEM`]: crate::errors::ENOMEM
186    pub fn send_msg<F>(&self, f: F) -> Result<()>
187    where
188        F: FnOnce(),
189    {
190        let ctx = Box::into_raw(Box::new(f));
191
192        unsafe {
193            to_result!(spdk_thread_send_msg(
194                self.as_ptr(),
195                Some(Self::handle_msg::<F>),
196                ctx.cast()
197            ))
198            .or_else(|e| {
199                drop(Box::from_raw(ctx));
200                Err(e)
201            })
202        }
203    }
204
205    /// Perform one iteration worth of processing on the thread.
206    ///
207    /// # Returns
208    ///
209    /// This method returns `true` if work was done and `false` otherwise.
210    pub(crate) fn poll(&self) -> bool {
211        unsafe { spdk_thread_poll(self.as_ptr() as *mut _, 0, 0) != 0 }
212    }
213
214    /// Spawns a new asynchronous task to be executed on this thread and returns a [`JoinHandle`] to
215    /// await results.
216    ///
217    /// The indirection of `fut_gen` instead of receiving a `Future` directly allows for futures
218    /// that may not be `Send` once started.
219    pub fn spawn<'a, G, F, R>(&'a self, fut_gen: G) -> JoinHandle<'a, F, R>
220    where
221        G: FnOnce() -> F + Send + 'a,
222        F: Future<Output = R> + 'a,
223        R: Send + 'a,
224    {
225        // SAFETY: Once scheduled, the task will only be accessed on the SPDK thread, ensuring that
226        // the unowned reference remains valid.
227        let task = RemoteTask::new(unsafe { self.as_unowned() }, fut_gen());
228
229        ArcTask::schedule_by_ref(&task);
230
231        JoinHandle::from_remote_task(task)
232    }
233
234    /// Spawns a new asynchronous task to be executed on this thread that will run to completion
235    /// independently of the current thread.
236    ///
237    /// The indirection of `fut_gen` instead of receiving a `Future` directly allows for futures
238    /// that may not be `Send` once started.
239    ///
240    /// # Safety
241    ///
242    /// This function is unsafe because it spawns a detached task that may outlive the calling
243    /// thread. The caller must ensure that calling thread remains alive for the duration of the
244    /// detached task coordinating by other means.
245    pub unsafe fn spawn_detached<G, F, R>(&self, fut_gen: G)
246    where
247        G: FnOnce() -> F + Send,
248        F: Future<Output = R>,
249        R: Send + 'static,
250    {
251        // SAFETY: Once scheduled, the task will only be accessed on the SPDK thread, ensuring that
252        // the unowned reference remains valid.
253        let task = RemoteTask::new(unsafe { self.as_unowned() }, fut_gen());
254
255        ArcTask::schedule(task);
256    }
257}

324/// Attempts to call the specified function object with a reference to the current [`Thread`] object.
325///
326/// # Return
327///
328/// If the current system thread is an SPDK thread, this function returns `Ok` with the result of
329/// the function object. Otherwise, it returns `Err(EINVAL)`.
330pub fn try_with_current<'a, F, R>(f: F) -> Result<R>
331where
332    F: FnOnce(&Thread<Unowned>) -> R + 'a,
333    R: 'a,
334{
335    // SAFTEY: The `Thread` instance is passed as a shared reference to the function object.
336    unsafe { Unowned::new(spdk_get_thread()) }
337        .map(Thread)
338        .map(|current| f(&current))
339        .ok_or(EINVAL)
340}
341
342/// Calls the specified function object with a reference to the current [`Thread`] object.
343///
344/// # Panics
345///
346/// This function panics if the current system thread is not an SPDK thread.
347pub fn with_current<'a, F, R>(f: F) -> R
348where
349    F: FnOnce(&Thread<Unowned>) -> R + 'a,
350    R: 'a,
351{
352    try_with_current(f).expect("called on SPDK thread")
353}

355/// Creates a new [`Thread`] to execute an asynchronous task and returns a [`JoinHandle`] to
356/// await results.
357///
358/// The indirection of `fut_gen` instead of receiving a `Future` directly allows for futures
359/// that may not be `Send` once started.
360pub fn spawn<'a, G, F, R>(name: &CStr, cpuset: &CpuSet, fut_gen: G) -> Result<JoinHandle<'a, F, R>>
361where
362    G: FnOnce() -> F + Send + 'a,
363    F: Future<Output = R> + 'a,
364    R: Send + 'a,
365{
366    let thread = Thread::new(name, cpuset)?;
367    let fut = fut_gen();
368
369    // SAFETY: The future owned by the task owns the SPDK thread ensuring that the unowned reference remains valid.
370    let task = RemoteTask::new(unsafe { thread.as_unowned() }, async move {
371        let res = fut.await;
372        drop(thread);
373        res
374    });
375
376    ArcTask::schedule_by_ref(&task);
377
378    Ok(JoinHandle::from_remote_task(task))
379}

381/// Creates a new [`Thread`] to execute an asynchronous task independently of the current task.
382///
383/// The indirection of `fut_gen` instead of receiving a `Future` directly allows for futures
384/// that may not be `Send` once started.
385pub fn spawn_detached<G, F, R>(name: &CStr, cpuset: &CpuSet, fut_gen: G) -> Result<()>
386where
387    G: FnOnce() -> F + Send,
388    F: Future<Output = R>,
389    R: Send + 'static,
390{
391    let thread = Thread::new(name, cpuset)?;
392    let fut = fut_gen();
393
394    // SAFETY: The future owned by the task owns the SPDK thread ensuring that the unowned reference remains valid.
395    let task = RemoteTask::new(unsafe { thread.as_unowned() }, async move {
396        let res = fut.await;
397        drop(thread);
398        res
399    });
400
401    ArcTask::schedule(task);
402
403    Ok(())
404}

406/// Spawns a new asynchronous task to be executed on the current SPDK thread and returns a
407/// [`JoinHandle`] to await results.
408pub fn spawn_local<'a, F, R>(fut: F) -> JoinHandle<'a, F, R>
409where
410    F: Future<Output = R> + 'a,
411    R: 'a,
412{
413    with_current(|current| {
414        // SAFETY: `JoinHandle` is `must_use` to ensure that the current thread remains valid for the
415        // duration of the task.
416        let task = LocalTask::new(unsafe { current.as_unowned() }, fut);
417
418        RcTask::schedule_by_ref(&task);
419
420        JoinHandle::from_local_task(task)
421    })
422}

424/// Spawns a new asynchronous task to be executed on the current SPDK thread that runs to completion
425/// independently of the current task.
426///
427/// # Safety
428///
429/// This function is unsafe because it spawns a detached task that may outlive the calling thread.
430/// The caller must ensure that calling thread remains alive for the duration of the detached task
431/// coordinating by other means.
432pub unsafe fn spawn_local_detached<F, R>(fut: F)
433where
434    F: Future<Output = R>,
435    R: 'static,
436{
437    with_current(|current| {
438        let task = LocalTask::new(unsafe { current.as_unowned() }, fut);
439
440        RcTask::schedule(task);
441    });
442}

444/// Runs the provided future on the current SPDK thread until completion.
445///
446/// # Notes
447///
448/// This function blocks the current reactor until the future completes on the current SPDK thread.
449/// Although the given future may spawn concurrent tasks on this thread, tasks on other threads
450/// associated with the current reactor will not run. The given future must not depend on the result
451/// of concurrent tasks associated with other threads, otherwise a deadlock will occur.
452pub fn block_on<'a, F, R>(fut: F) -> R
453where
454    F: Future<Output = R> + 'a,
455    R: 'static,
456{
457    let mut join_handle = spawn_local(fut);
458
459    with_current(|current_thread| {
460        loop {
461            if let Poll::Ready(res) =
462                Pin::new(&mut join_handle).poll(&mut Context::from_waker(noop_waker_ref()))
463            {
464                return res;
465            }
466
467            current_thread.poll();
468        }
469    })
