31/// The type of NVMe-oF transport.
32///
33/// # Notes
34///
35/// These are mapped directly to the NVMe over Fabrics TRTYPE values, except for PCIe, which is a
36/// special case since NVMe over Fabrics does not define a TRTYPE for local PCIe.
37///
38/// Transports supported by SPDK but not defined in the NVMe-oF specification are given values
39/// outside of the 8-bit range of the TRTYPE value.
40#[derive(Clone, Copy, Debug, Eq, PartialEq)]
41pub enum TransportType {
42    /// A RDMA Transport.
43    RDMA = 1,
44
45    /// A Fibre Channel transport.
46    FC = 2,
47
48    /// A TCP Transport.
49    TCP = 3,
50
51    /// A PCIe Transport.
52    PCIE = 256,
53
54    /// A user-mode vfio transport.
55    #[cfg(feature = "nvme-vfio-user")]
56    VFIOUSER = 1024,
57
58    /// A custom transport.
59    CUSTOM = 4096,
60}
61
62impl From<TransportType> for &'static CStr {
63    fn from(transport_type: TransportType) -> Self {
64        match transport_type {
65            TransportType::CUSTOM => SPDK_NVME_TRANSPORT_NAME_CUSTOM,
66            TransportType::FC => SPDK_NVME_TRANSPORT_NAME_FC,
67            TransportType::PCIE => SPDK_NVME_TRANSPORT_NAME_PCIE,
68            TransportType::RDMA => SPDK_NVME_TRANSPORT_NAME_RDMA,
69            TransportType::TCP => SPDK_NVME_TRANSPORT_NAME_TCP,
70
71            #[cfg(feature = "nvme-vfio-user")]
72            TransportType::VFIOUSER => SPDK_NVME_TRANSPORT_NAME_VFIOUSER,
73        }
74    }
75}

153                    }

242/// Represents the ownership state of a [`Transport`].
243#[derive(Debug)]
244enum OwnershipState {
245    Owned(NonNull<spdk_nvmf_transport>),
246    Borrowed(NonNull<spdk_nvmf_transport>),
247    None,
248}
249
250unsafe impl Send for OwnershipState {}
251
252/// Represents a NVMe-oF transport.
253///
254/// `Transport` wraps an `spdk_nvmf_transport` pointer and can be in one of three ownership states:
255/// owned, borrowed, or none.
256///
257/// An owned transport owns the underlying `spdk_nvmf_transport` pointer and will destroy it when
258/// dropped. The caller must ensure that the drop occurs in the same thread that created the
259/// transport. It must also occur as part of thread event handling by explicitly calling
260/// [`task::yield_now`] before dropping the transport. However, it is easiest and safest to
261/// explicitly call [`Transport::destroy`] on the transport rather than let it drop naturally.
262///
263/// A borrowed transport borrows the underlying `spdk_nvmf_transport` pointer. Dropping a borrowed
264/// transport has no effect on the underlying `spdk_nvmf_transport` pointer.
265///
266/// A transport with no ownership state can only be safely queried for ownership state or dropped.
267/// Any other operation will panic. A transport will be left in this state after the
268/// [`Transport::take`] method is called.
269///
270/// [`Transport::destroy`]: method@Transport::destroy
271/// [`Transport::take`]: method@Transport::take
272/// [`task::yield_now`]: function@crate::task::yield_now
273#[derive(Debug)]
274pub struct Transport(OwnershipState);
275
276unsafe impl Send for Transport {}
277
278impl Transport {
279    /// Returns a borrowed transport from a raw `spdk_nvmf_transport` pointer.
280    pub(crate) fn from_ptr(ptr: *const spdk_nvmf_transport) -> Self {
281        Self::try_from_ptr(ptr).expect("transport pointer must not be null")
282    }
283
284    /// Attempts to return a borrowed transport from a raw `spdk_nvmf_transport` pointer.
285    ///
286    /// This function returns `None` if the pointer was null.
287    pub(crate) fn try_from_ptr(ptr: *const spdk_nvmf_transport) -> Option<Self> {
288        NonNull::new(ptr as *mut spdk_nvmf_transport).map(|ptr| Self(OwnershipState::Borrowed(ptr)))
289    }
290
291    /// Returns a pointer to the underlying `spdk_nvmf_transport` structure.
292    pub fn as_ptr(&self) -> *mut spdk_nvmf_transport {
293        match &self.0 {
294            OwnershipState::Owned(ptr) | OwnershipState::Borrowed(ptr) => ptr.as_ptr(),
295            OwnershipState::None => panic!("no transport"),
296        }
297    }
298
299    /// Borrows the transport.
300    pub fn borrow(&self) -> Self {
301        match self.0 {
302            OwnershipState::Owned(ptr) | OwnershipState::Borrowed(ptr) => {
303                Self(OwnershipState::Borrowed(ptr))
304            }
305            OwnershipState::None => panic!("no transport"),
306        }
307    }
308
309    /// Returns whether the transport is owned.
310    pub fn is_owned(&self) -> bool {
311        matches!(self.0, OwnershipState::Owned(_))
312    }
313
314    /// Returns whether the transport is borrowed.
315    pub fn is_borrowed(&self) -> bool {
316        matches!(self.0, OwnershipState::Borrowed(_))
317    }
318
319    /// Returns whether this transport has no ownership state.
320    pub fn is_none(&self) -> bool {
321        matches!(self.0, OwnershipState::None)
322    }
323
324    /// Takes the value from this transport and replaces with a value having no ownership.
325    pub fn take(&mut self) -> Self {
326        mem::replace(self, Self(OwnershipState::None))
327    }
328
329    /// Returns the name of the transport.
330    pub fn name(&self) -> &'static CStr {
331        unsafe { CStr::from_ptr(spdk_nvmf_get_transport_name(self.as_ptr())) }
332    }
333
334    /// Returns the type of the transport.
335    pub fn r#type(&self) -> TransportType {
336        unsafe {
337            spdk_nvmf_get_transport_type(self.as_ptr())
338                .try_into()
339                .expect("valid transport type")
340        }
341    }
342
343    /// Destroys the transport asynchronously.
344    ///
345    /// # Returns
346    ///
347    /// Only an owned transport can be destroyed. This function returns `Err(EPERM)` if called on a
348    /// borrowed transport and `Err(ENODEV)` if called on a transport that neither owns nor borrows
349    /// the underlying `spdk_nvmf_transport` pointer.
350    pub async fn destroy(mut self) -> Result<()> {
351        match self.0 {
352            OwnershipState::Borrowed(_) => Err(EPERM),
353            OwnershipState::None => Err(ENODEV),
354            OwnershipState::Owned(_) => {
355                Promise::new()
356                    .request(move |p| {
357                        let (cb_fn, cb_arg) = Promissory::callback_with_ok(p);
358
359                        let res = to_poll_pending_on_ok! {
360                            unsafe {
361                                spdk_nvmf_transport_destroy(
362                                    self.as_ptr(),
363                                    Some(cb_fn),
364                                    cb_arg.cast_mut() as *mut _,
365                                )
366                            }
367                            => on ready {
368                                unsafe { drop(Promissory::from_raw(cb_arg)) };
369                            }
370                        };
371
372                        mem::forget(self.take());
373
374                        res
375                    })
376                    .await
377            }
378        }
379    }
380}

382impl Drop for Transport {
383    fn drop(&mut self) {
384        if self.is_owned() {
385            let transport = self.take();
386
387            thread::block_on(async move { transport.destroy().await }).unwrap();
388        }
389    }
390}

425impl Iterator for Transports {
426    type Item = Transport;
427
428    fn next(&mut self) -> Option<Self::Item> {
429        if self.0.is_null() {
430            return None;
431        }
432        let transport = self.0;

434        self.0 = unsafe { spdk_nvmf_transport_get_next(self.0) };
